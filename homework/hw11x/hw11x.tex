\begin{remember}[The deepest water]
  \noindent\begin{minipage}[c]{0.66\linewidth}
  The water goes \key{half the range above} the average depth, and \key{half the
  range below} it. \\[3pt]
  Average $5.0$ m, \, range $2.0$ m: \; half the range is $1.0$ m. \\
